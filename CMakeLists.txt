cmake_minimum_required(VERSION 3.25)
project(BattleBoundDX LANGUAGES CXX)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR OR CMAKE_BINARY_DIR MATCHES "^${CMAKE_SOURCE_DIR}/")
  message(FATAL_ERROR "Use tools/build.ps1: build outputs must stay outside the workspace.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/workspace_budget.py" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/generate_content.py" COMMAND_ERROR_IS_FATAL ANY)
find_path(RAYLIB_INCLUDE_DIR raylib.h REQUIRED)
find_file(RAYLIB_LIBRARY NAMES libraylib.a PATHS "C:/msys64/ucrt64/lib" REQUIRED)
find_file(GLFW_LIBRARY NAMES libglfw3.a PATHS "C:/msys64/ucrt64/lib" REQUIRED)
find_path(VULKAN_HEADERS vulkan/vulkan.h HINTS "$ENV{VULKAN_SDK}/Include" REQUIRED)
find_program(GLSLC glslc HINTS "$ENV{VULKAN_SDK}/Bin" REQUIRED)
find_program(SPIRV_VAL spirv-val HINTS "$ENV{VULKAN_SDK}/Bin" REQUIRED)
execute_process(COMMAND "${GLSLC}" "${CMAKE_SOURCE_DIR}/src/shaders/sprite.vert" -o "${CMAKE_BINARY_DIR}/sprite.vert.spv" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${GLSLC}" "${CMAKE_SOURCE_DIR}/src/shaders/sprite.frag" -o "${CMAKE_BINARY_DIR}/sprite.frag.spv" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${SPIRV_VAL}" --target-env vulkan1.0 "${CMAKE_BINARY_DIR}/sprite.vert.spv" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${SPIRV_VAL}" --target-env vulkan1.0 "${CMAKE_BINARY_DIR}/sprite.frag.spv" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/embed_shaders.py" "${CMAKE_BINARY_DIR}/sprite.vert.spv" "${CMAKE_BINARY_DIR}/sprite.frag.spv" "${CMAKE_BINARY_DIR}/sprite_shaders.hpp" COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/static_glfw.py" "${RAYLIB_LIBRARY}" "${CMAKE_BINARY_DIR}/static_glfw.cpp" COMMAND_ERROR_IS_FATAL ANY)
add_executable(battle_bound_dx WIN32 src/main.cpp src/runtime.cpp src/movement.cpp src/combat.cpp src/weapon_actions.cpp src/combat_check.cpp src/multiplayer.cpp src/lobby.cpp src/network_combat.cpp src/game_ui.cpp src/assets.cpp src/graphics.cpp src/content.cpp src/replay.cpp src/input.cpp src/platform_input.cpp src/camera.cpp src/camera_bridge.cpp "${CMAKE_BINARY_DIR}/static_glfw.cpp")
target_compile_options(battle_bound_dx PRIVATE -std=c++2c -Wall -Wextra -Wpedantic -Werror)
target_include_directories(battle_bound_dx PRIVATE ${RAYLIB_INCLUDE_DIR} ${VULKAN_HEADERS} "${CMAKE_BINARY_DIR}")
target_link_libraries(battle_bound_dx PRIVATE ${RAYLIB_LIBRARY} ${GLFW_LIBRARY})
if(WIN32)
  target_link_options(battle_bound_dx PRIVATE -static -static-libgcc -static-libstdc++)
  target_link_libraries(battle_bound_dx PRIVATE winmm gdi32 opengl32 user32 shell32 ws2_32)
endif()
enable_testing()
add_test(NAME native_self_test COMMAND battle_bound_dx --self-test)
add_test(NAME original_project_smoke COMMAND battle_bound_dx --headless --ticks 300 --root "${CMAKE_SOURCE_DIR}")
add_test(NAME offline_game_smoke COMMAND battle_bound_dx --headless --game --ticks 300 --root "${CMAKE_SOURCE_DIR}")
add_test(NAME movement_combat_smoke COMMAND battle_bound_dx --headless --exercise --ticks 300 --root "${CMAKE_SOURCE_DIR}")
add_test(NAME deterministic_replay COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/test_replay.py" "$<TARGET_FILE:battle_bound_dx>")
add_test(NAME asset_provenance COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/verify_assets.py")
add_test(NAME input_camera_parity COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/test_input_camera.py" "$<TARGET_FILE:battle_bound_dx>")
set_tests_properties(input_camera_parity PROPERTIES TIMEOUT 150)
set_tests_properties(deterministic_replay PROPERTIES TIMEOUT 150)
set_tests_properties(native_self_test original_project_smoke offline_game_smoke movement_combat_smoke PROPERTIES TIMEOUT 60)
add_test(NAME combat_replay COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/test_combat.py" "$<TARGET_FILE:battle_bound_dx>")
set_tests_properties(combat_replay PROPERTIES TIMEOUT 125)
add_test(NAME opponent_combat COMMAND battle_bound_dx --headless --opponent-check --root "${CMAKE_SOURCE_DIR}")
add_test(NAME arsenal_combat COMMAND battle_bound_dx --headless --arsenal-check --root "${CMAKE_SOURCE_DIR}")
set_tests_properties(opponent_combat arsenal_combat PROPERTIES TIMEOUT 150)

option(BBDX_NETWORK_TESTS "Enable the two-client loopback test" OFF)
if(BBDX_NETWORK_TESTS)
  add_test(NAME multiplayer_combat COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/test_multiplayer.py" "$<TARGET_FILE:battle_bound_dx>")
  set_tests_properties(multiplayer_combat PROPERTIES TIMEOUT 150)
endif()
